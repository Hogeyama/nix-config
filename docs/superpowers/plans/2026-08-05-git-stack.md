# `git stack` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `git stack` command that edits an ancestor stack branch, then rebases the original branch and updates intermediate branch refs.

**Architecture:** A standalone Bash `git-stack` executable owns graph selection, per-worktree state, and the `edit`/`done` state machine. A plain Bash integration suite creates isolated repositories and exercises the executable through Git, while the navi cheat delegates branch completion to `git stack list`.

**Tech Stack:** Bash, Git 2.43-compatible porcelain and plumbing commands, navi cheat syntax, shellcheck, Nix flake evaluation.

## Global Constraints

- Store edit state under the path returned by `git rev-parse --git-path stack-edit`.
- Accept multiple `stack.baseBranch` values; if unset, try local `main` and `master`.
- Select the applicable base with the shortest ancestry path, preserving configured order as the tie-break.
- Require a clean worktree for `edit` and the initial `done`, but allow staged resolutions for rebase continuation.
- Use `git rebase --onto <target> <old-target-oid> --update-refs`; do not replay the old target commit.
- Provide no custom abort subcommand and never rewind commits made on the edited branch.
- Keep graph logic in `git-stack`; navi completion must call `git stack list`.

#!/usr/bin/env bash
set -euo pipefail

usage() {
    cat <<'EOF'
NAME
       git-stack - Edit a branch in a stack

SYNOPSIS
       git stack edit <local-branch>
       git stack done
       git stack list
EOF
}

die() {
    echo "git stack: $*" >&2
    exit 1
}

require_worktree() {
    git rev-parse --is-inside-work-tree >/dev/null 2>&1 ||
        die "not inside a Git worktree"
}

current_branch() {
    git symbolic-ref --quiet --short HEAD
}

local_branch_exists() {
    git show-ref --verify --quiet "refs/heads/$1"
}

select_base() {
    local current=$1
    local current_ref=refs/heads/$current
    local candidate candidate_ref distance
    local best=
    local best_distance=
    local -a candidates=()

    mapfile -t candidates < <(git config --get-all stack.baseBranch || :)
    if ((${#candidates[@]} == 0)); then
        candidates=(main master)
    fi

    for candidate in "${candidates[@]}"; do
        candidate_ref=refs/heads/$candidate
        local_branch_exists "$candidate" || continue
        git merge-base --is-ancestor "$candidate_ref" "$current_ref" || continue
        distance=$(git rev-list --ancestry-path --count "$candidate_ref..$current_ref")
        if [[ -z $best_distance ]] || ((distance < best_distance)); then
            best=$candidate
            best_distance=$distance
        fi
    done

    [[ -n $best ]] ||
        die "no base branch applies; configure one with: git config --add stack.baseBranch <branch>"
    printf '%s\n' "$best"
}

stack_branches() {
    local current=$1
    local base current_oid oid
    base=$(select_base "$current")
    current_oid=$(git rev-parse "refs/heads/$current")

    while IFS= read -r oid; do
        [[ $oid == "$current_oid" ]] && continue
        git for-each-ref \
            --points-at="$oid" \
            --sort=refname \
            --format='%(refname:short)' \
            refs/heads
    done < <(
        git rev-list \
            --topo-order \
            --reverse \
            --ancestry-path \
            "refs/heads/$base..refs/heads/$current"
    )
}

command_list() {
    local current
    current=$(current_branch) || die "detached HEAD is not a stack branch"
    stack_branches "$current"
}

main() {
    case ${1:-} in
        -h | --help)
            (($# == 1)) || die "--help takes no arguments"
            usage
            ;;
        list)
            (($# == 1)) || die "usage: git stack list"
            require_worktree
            command_list
            ;;
        "")
            usage >&2
            exit 1
            ;;
        *)
            die "unknown subcommand: $1"
            ;;
    esac
}

main "$@"

command_edit() {
    local target=$1
    local current old_target_oid candidate
    local eligible=false

    [[ ! -d $state_dir ]] ||
        die "another stack edit is already active"
    require_clean_worktree
    current=$(current_branch) || die "detached HEAD cannot start a stack edit"
    local_branch_exists "$target" ||
        die "local branch does not exist: $target"
    [[ $target != "$current" ]] ||
        die "cannot edit the current branch: $target"

    while IFS= read -r candidate; do
        if [[ $candidate == "$target" ]]; then
            eligible=true
            break
        fi
    done < <(stack_branches "$current")
    [[ $eligible == true ]] ||
        die "$target is not editable from $current"

    old_target_oid=$(git rev-parse "refs/heads/$target")
    write_edit_state "$target" "$old_target_oid" "$current"
    if ! git switch "$target"; then
        clear_state
        die "could not switch to $target"
    fi
}

    [[ -d $(git rev-parse --git-path rebase-merge) ||

load_state() {
    local file
    [[ -d $state_dir ]] || die "no stack edit is active"
    for file in target-branch old-target-oid return-branch phase; do
        [[ -f $state_dir/$file ]] ||
            die "stack edit state is incomplete: $state_dir"

    target_branch=$(<"$state_dir/target-branch")
    old_target_oid=$(<"$state_dir/old-target-oid")
    return_branch=$(<"$state_dir/return-branch")

    [[ $phase == edit || $phase == rebase ]] ||
        die "unknown stack edit phase: $phase"
    git cat-file -e "$old_target_oid^{commit}" 2>/dev/null ||
        die "old target commit is unavailable: $old_target_oid"

command_done() {
    local current
    load_state

    if [[ $phase == rebase ]]; then
        rebase_in_progress ||
            die "recorded rebase is no longer in progress"
        if git rebase --continue; then
            clear_state
            return 0
        fi
        rebase_in_progress || clear_state
        return 1
    fi

    require_clean_worktree
    current=$(current_branch) ||
        die "target branch is not checked out"
    [[ $current == "$target_branch" ]] ||
        die "expected $target_branch, found $current"
    local_branch_exists "$return_branch" ||
        die "return branch no longer exists: $return_branch"

    git switch "$return_branch"
    printf '%s\n' rebase >"$state_dir/phase"
    if git rebase \
        --onto "refs/heads/$target_branch" \
        "$old_target_oid" \
        --update-refs; then
        clear_state
        return 0
    fi
    rebase_in_progress || clear_state
    return 1
}
